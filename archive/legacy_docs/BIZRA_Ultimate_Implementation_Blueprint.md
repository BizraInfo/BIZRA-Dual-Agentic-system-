    """
    
    PHI = 1.618
    RETENTION_RATIO = 1 / PHI  # ≈ 0.618
    FIBONACCI_STEPS = [13, 21, 34, 55, 89, 144, 233, 377]

        # L-COLD: PostgreSQL with pgvector
        self.L_COLD_backend = None  # PostgreSQL connection
        
        # Counters for Fibonacci condensation
        self.step_counter = 0
    
    def store(self, content: Any, layer: str = "HOT", metadata: Dict = None) -> str:
        """Store content in specified layer."""
        
        entry = MemoryEntry(
            content=content,
            metadata=metadata or {},
            timestamp=datetime.now(timezone.utc).isoformat()
        )
        
        key = entry.hash()
        
        if layer == "HOT":
            self.L_HOT[key] = entry
        elif layer == "WARM":
            self._store_warm(key, entry)
        else:  # COLD
            self._store_cold(key, entry)

            return self.L_HOT[key]

```python
class ZeroTrustGate:
    """
    Fortress-class security gate.
    
    Every request passes through:
    1. Input sanitization (prompt injection detection)
    2. Intent classification (risk scoring)
    3. Privilege verification (least privilege)
    4. Cryptographic attestation (TPM/HSM)
    """
    
    FORBIDDEN_PATTERNS = [
        r"ignore.*previous.*instructions",
        r"disregard.*all",
        r"system\s*prompt",
        r"jailbreak",
        r"pretend.*you.*are",
        r"roleplay.*as",
        r"bypass.*safety",
        r"reveal.*instructions"
    ]
    
    def __init__(self):
        self.attestation_chain: List[Attestation] = []
        self.threat_model = "FORTRESS"
        self.tpm_available = self._check_tpm()
    
    def verify_request(self, request: Dict) -> Tuple[bool, str, Optional[Attestation]]:
        """
        Zero Trust verification pipeline.
        
        Returns: (passed, message, attestation)
        """
        
        # Stage 1: Input Sanitization
        sanitize_result = self._sanitize_input(request)
        if not sanitize_result.passed:
            return (False, f"BLOCKED: {sanitize_result.reason}", None)
        
        # Stage 2: Intent Classification
        intent = self._classify_intent(request)
        if intent.risk_score > 0.7:
            return (False, f"BLOCKED: High-risk intent ({intent.risk_score:.2f})", None)
        
        # Stage 3: Privilege Check
        priv_result = self._check_privilege(request)
        if not priv_result.authorized:
            return (False, f"BLOCKED: Insufficient privilege for {priv_result.required_scope}", None)
        
        # Stage 4: Cryptographic Attestation
        attestation = self._generate_attestation(request)
        self.attestation_chain.append(attestation)
        
        return (True, f"VERIFIED: {attestation.hash[:16]}", attestation)
    
    def _sanitize_input(self, request: Dict) -> SanitizeResult:
        """Detect and block injection attempts."""
        
        content = json.dumps(request).lower()
        
        for pattern in self.FORBIDDEN_PATTERNS:
            if re.search(pattern, content):
                return SanitizeResult(
                    passed=False,
                    reason=f"Injection pattern detected: {pattern}"
                )
        
        return SanitizeResult(passed=True, reason="Clean")
    
    def _classify_intent(self, request: Dict) -> IntentClassification:
        """Classify request intent and compute risk score."""
        
        # Risk factors
        risk_score = 0.0
        
        # Large payload = higher risk
        payload_size = len(json.dumps(request))
        if payload_size > 10000:
            risk_score += 0.2
        
        # Code execution = higher risk
        if "code" in request or "exec" in request:
            risk_score += 0.3
        
        # File system access = higher risk
        if "file" in request or "path" in request:
            risk_score += 0.2
        
        # Network access = higher risk
        if "url" in request or "http" in request:
            risk_score += 0.2
        
        return IntentClassification(
            intent_type=request.get("type", "UNKNOWN"),
            risk_score=min(1.0, risk_score),
            factors=[]
        )
    
    def _generate_attestation(self, request: Dict) -> Attestation:
        """Generate cryptographic attestation."""
        
        timestamp = datetime.now(timezone.utc).isoformat()
        
        payload = {
            "request_hash": hashlib.sha256(json.dumps(request).encode()).hexdigest(),
            "timestamp": timestamp,
            "threat_model": self.threat_model,
            "tpm_attested": self.tpm_available
        }
        
        attestation_hash = hashlib.sha256(
            json.dumps(payload, sort_keys=True).encode()
        ).hexdigest()
        
        return Attestation(
            hash=attestation_hash,
            timestamp=timestamp,
            payload=payload,
            tpm_signature=self._tpm_sign(attestation_hash) if self.tpm_available else None
        )
```
